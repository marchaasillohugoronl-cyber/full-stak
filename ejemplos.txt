1. Sumar elementos de un array

javascript
let numeros = [1, 2, 3, 4, 5];
let suma = 0;

for (let i = 0; i < numeros.length; i++) {
    suma += numeros[i];
}

console.log("La suma es:", suma);
2. Encontrar el número mayor y menor

javascript
let numeros = [8, 3, 10, 1, 6];
let mayor = numeros[0];
let menor = numeros[0];

for (let i = 1; i < numeros.length; i++) {
    if (numeros[i] > mayor) mayor = numeros[i];

}

console.log("Mayor:", mayor);
console.log("Menor:", menor);

4. Ordenar un array sin usar sort()

javascript
let numeros = [5, 2, 4, 1];

for (let i = 0; i < numeros.length; i++) {
    for (let j = 0; j < numeros.length - 1; j++) {
        if (numeros[j] > numeros[j + 1]) {
            let temp = numeros[j];
            numeros[j] = numeros[j + 1];
            numeros[j + 1] = temp;
        }
    }
}

console.log(numeros);

javascript

17. Contador de intentos de login

javascript
const prompt = require("prompt-sync")();

let claveCorrecta = "1234";
let clave = "";
let intentos = 0;

while (clave !== claveCorrecta && intentos < 3) {
    clave = prompt("Ingresa la contraseña: ");
    intentos++;
}

if (clave === claveCorrecta) {
    console.log("Acceso permitido");
} else {
    console.log("Acceso bloqueado");
}
